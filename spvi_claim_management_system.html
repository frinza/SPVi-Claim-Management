1

2
ไปที่ Extensions → Apps Script แล้วสร้าง Web App จาก doPost script (ดูตัวอย่างด้านล่าง)
3
Deploy → New deployment → Web App → Execute as: Me, Access: Anyone → Copy URL
4
วาง URL ด้านล่าง แล้วกด บันทึก

function doPost(e) { var lock = LockService.getDocumentLock(); lock.waitLock(30000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('ClaimLog') || ss.getSheets()[0]; var payload = JSON.parse(e.postData.contents || '[]'); var rows = Array.isArray(payload) ? payload : [payload]; if (rows.length === 0) { return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: 0 })) .setMimeType(ContentService.MimeType.JSON); } var action = rows[0]['_action'] || 'append'; rows = rows.map(function(r) { var c = {}; Object.keys(r).forEach(function(k) { if (k !== '_action') c[k] = r[k]; }); return c; }); if (action === 'updateStatus' && sheet.getLastRow() > 1) { var lastCol = sheet.getLastColumn(); var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; var claimNoIdx = headers.indexOf('เลขที่เอกสาร'); var seqIdx = headers.indexOf('ลำดับสินค้า'); var statusRowIdx = headers.indexOf('สถานะรายการ'); var statusDocIdx = headers.indexOf('สถานะเอกสาร'); var savedAtIdx = headers.indexOf('บันทึกเมื่อ'); var typeIdx = headers.indexOf('ประเภทบันทึก'); var approvedQtyIdx = headers.indexOf('จำนวนอนุมัติ'); var bookingNoIdx = headers.indexOf('Booking No.'); var cnNumberIdx = headers.indexOf('CN Number'); var claimUpdatedAtIdx = ensureCol_(sheet, headers, 'อัพเดทเมื่อ'); var completedAtIdx = ensureCol_(sheet, headers, 'เสร็จสิ้นเมื่อ'); var bookingDateIdx = ensureCol_(sheet, headers, 'วันที่ Booking'); var cnDateIdx = ensureCol_(sheet, headers, 'วันที่ CN'); var dateDiffCnIdx = ensureCol_(sheet, headers, 'จำนวนวัน Booking→CN'); var returnNoIdx = ensureCol_(sheet, headers, 'เลขที่โอนคืนคลัง'); var returnDateIdx = ensureCol_(sheet, headers, 'วันที่คืนสินค้า'); var adjustNoIdx = ensureCol_(sheet, headers, 'เลขที่ Adjust'); var adjustDateIdx = ensureCol_(sheet, headers, 'วันที่ Adjust'); var statusNoteIdx = ensureCol_(sheet, headers, 'เหตุผลสถานะ'); var itemNoteIdx = headers.indexOf('หมายเหตุสินค้า'); if (claimNoIdx > -1 && seqIdx > -1 && statusRowIdx > -1) { var lastRow = sheet.getLastRow(); var dataVals = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); var notFound = []; var updated = 0; rows.forEach(function(row) { var found = false; for (var i = 0; i < dataVals.length; i++) { if (String(dataVals[i][claimNoIdx]) === String(row['เลขที่เอกสาร']) && String(dataVals[i][seqIdx]) === String(row['ลำดับสินค้า'])) { var sheetRow = i + 2; sheet.getRange(sheetRow, statusRowIdx + 1).setValue(row['สถานะรายการ'] || ''); if (statusDocIdx > -1) sheet.getRange(sheetRow, statusDocIdx + 1).setValue(row['สถานะเอกสาร'] || ''); if (typeIdx > -1) sheet.getRange(sheetRow, typeIdx + 1).setValue('อัปเดตสถานะ'); if (approvedQtyIdx > -1) sheet.getRange(sheetRow, approvedQtyIdx + 1).setValue(row['จำนวนอนุมัติ'] != null ? row['จำนวนอนุมัติ'] : ''); if (bookingNoIdx > -1) sheet.getRange(sheetRow, bookingNoIdx + 1).setValue(row['Booking No.'] != null ? row['Booking No.'] : ''); if (cnNumberIdx > -1) sheet.getRange(sheetRow, cnNumberIdx + 1).setValue(row['CN Number'] != null ? row['CN Number'] : ''); if (claimUpdatedAtIdx > -1) sheet.getRange(sheetRow, claimUpdatedAtIdx + 1).setValue(row['อัพเดทเมื่อ'] || ''); if (completedAtIdx > -1) sheet.getRange(sheetRow, completedAtIdx + 1).setValue(row['เสร็จสิ้นเมื่อ'] || ''); if (bookingDateIdx > -1) sheet.getRange(sheetRow, bookingDateIdx + 1).setValue(row['วันที่ Booking'] || ''); if (cnDateIdx > -1) sheet.getRange(sheetRow, cnDateIdx + 1).setValue(row['วันที่ CN'] || ''); if (dateDiffCnIdx > -1) sheet.getRange(sheetRow, dateDiffCnIdx + 1).setValue(row['จำนวนวัน Booking→CN'] != null ? row['จำนวนวัน Booking→CN'] : ''); if (returnNoIdx > -1) sheet.getRange(sheetRow, returnNoIdx + 1).setValue(row['เลขที่โอนคืนคลัง'] || ''); if (returnDateIdx > -1) sheet.getRange(sheetRow, returnDateIdx + 1).setValue(row['วันที่คืนสินค้า'] || ''); if (adjustNoIdx > -1) sheet.getRange(sheetRow, adjustNoIdx + 1).setValue(row['เลขที่ Adjust'] || ''); if (adjustDateIdx > -1) sheet.getRange(sheetRow, adjustDateIdx + 1).setValue(row['วันที่ Adjust'] || ''); if (statusNoteIdx > -1) sheet.getRange(sheetRow, statusNoteIdx + 1).setValue(row['เหตุผลสถานะ'] || ''); if (itemNoteIdx > -1) sheet.getRange(sheetRow, itemNoteIdx + 1).setValue(row['หมายเหตุสินค้า'] != null ? row['หมายเหตุสินค้า'] : ''); updated++; found = true; break; } } if (!found) notFound.push(row); }); if (notFound.length) appendRows_(sheet, notFound); return ContentService.createTextOutput( JSON.stringify({ status: 'ok', updated: updated, appended: notFound.length }) ).setMimeType(ContentService.MimeType.JSON); } } return appendRows_(sheet, rows); } finally { lock.releaseLock(); } } function ensureCol_(sheet, headers, colName) { var idx = headers.indexOf(colName); if (idx === -1) { idx = headers.length; sheet.getRange(1, idx + 1).setValue(colName); headers.push(colName); } return idx; } function appendRows_(sheet, rows) { var headers = Object.keys(rows[0]); var firstRow = sheet.getRange(1, 1, 1, headers.length).getValues()[0]; var hasSameHeader = headers.every(function(h, i) { return String(firstRow[i] || '').trim() === h; }); if (!hasSameHeader) { if (sheet.getLastRow() === 0) { sheet.appendRow(headers); } else { sheet.insertRows(1, 1); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); } } var values = rows.map(function(r) { return headers.map(function(h) { return r[h] == null ? '' : r[h]; }); }); sheet.getRange(sheet.getLastRow() + 1, 1, values.length, headers.length).setValues(values); return ContentService.createTextOutput( JSON.stringify({ status: 'ok', count: values.length }) ).setMimeType(ContentService.MimeType.JSON); } function serializeSheetValue_(header, value, tz) { if (!(value instanceof Date) || isNaN(value)) return value; var dateOnlyColumns = { 'วันที่': true, }; if (dateOnlyColumns[String(header)]) { return Utilities.formatDate(value, tz, 'yyyy-MM-dd'); } return Utilities.formatDate(value, tz, "yyyy-MM-dd'T'HH:mm:ssXXX"); } function serializeRow_(headers, row, tz) { var obj = {}; headers.forEach(function(h, i) { obj[String(h)] = serializeSheetValue_(h, row[i], tz); }); return obj; } function doGet(e) { var params = (e && e.parameter) ? e.parameter : {}; var ss = SpreadsheetApp.getActiveSpreadsheet(); var tz = ss.getSpreadsheetTimeZone() || Session.getScriptTimeZone() || 'Asia/Bangkok'; if (params.sheet === 'Vender') { var vSheet = ss.getSheetByName('Vender'); if (!vSheet || vSheet.getLastRow() < 2) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var vLastCol = vSheet.getLastColumn(); var vHeaders = vSheet.getRange(1, 1, 1, vLastCol).getValues()[0]; var vData = vSheet.getRange(2, 1, vSheet.getLastRow() - 1, vLastCol).getValues(); var vRows = vData .filter(function(r) { return String(r[0]).trim() !== ''; }) .map(function(row) { return serializeRow_(vHeaders, row, tz); }); return ContentService.createTextOutput(JSON.stringify(vRows)) .setMimeType(ContentService.MimeType.JSON); } var sheet = ss.getSheetByName('ClaimLog') || ss.getSheets()[0]; var lastRow = sheet.getLastRow(); if (lastRow < 2) { return ContentService.createTextOutput(JSON.stringify([])) .setMimeType(ContentService.MimeType.JSON); } var lastCol = sheet.getLastColumn(); var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; var data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); var rows = data.map(function(row) { return serializeRow_(headers, row, tz); }); return ContentService.createTextOutput(JSON.stringify(rows)) .setMimeType(ContentService.MimeType.JSON); }
